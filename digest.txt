def analyze_content(content, guidelines):

    issues = []
    suggestions = []

    content_lower = content.lower()

    # Get restricted words from the organisation's guidelines
    prohibited_words = guidelines.get("restricted_words", [])

    for word in prohibited_words:

        if word.lower() in content_lower:

            issues.append(
                {
                    "type": "Terminology",
                    "severity": "High",
                    "message": f"Prohibited phrase detected: '{word}'",
                    "evidence": f"'{word}' is restricted by the brand guidelines.",
                }
            )

            suggestions.append(
                {
                    "original": word,
                    "replacement": "Use an approved, verifiable statement.",
                }
            )

    return {
        "issues": issues,
        "suggestions": suggestions,
        "overall_status": "Needs Review" if issues else "Passed Initial Checks",
    }

import re


def analyze_content(content, guidelines):

    issues = []
    suggestions = []

    restricted_words = guidelines.get("restricted_words", [])
    approved_terms = guidelines.get("approved_terms", [])
    approved_facts = guidelines.get("approved_facts", {})
    tone = guidelines.get("tone", "")

    # 1. Check restricted words and phrases
    for word in restricted_words:

        if not isinstance(word, str) or not word.strip():
            continue

        word = word.strip()

        pattern = rf"\b{re.escape(word)}\b"

        for match in re.finditer(pattern, content, re.IGNORECASE):

            detected_word = match.group()

            issues.append({
                "type": "Terminology",
                "severity": "High",
                "message": f"Restricted phrase detected: '{detected_word}'",
                "evidence": (
                    f"'{word}' is restricted by the brand guidelines."
                )
            })

            suggestions.append({
                "original": detected_word,
                "replacement": "Use an approved, verifiable statement."
            })

    # 2. Check approved terminology
    if isinstance(approved_terms, list):

        for term in approved_terms:

            if not isinstance(term, str) or not term.strip():
                continue

            pattern = rf"\b{re.escape(term.strip())}\b"

            if not re.search(pattern, content, re.IGNORECASE):
                issues.append({
                    "type": "Terminology",
                    "severity": "Low",
                    "message": f"Approved brand term '{term}' was not found.",
                    "evidence": (
                        f"Consider using the approved term '{term}' "
                        "where relevant."
                    )
                })

                suggestions.append({
                    "original": "",
                    "replacement": (
                        f"Consider using '{term}' where relevant."
                    )
                })

    # 3. Check approved facts
    if isinstance(approved_facts, dict):

        for subject, fact in approved_facts.items():

            if not isinstance(subject, str):
                continue

            if not isinstance(fact, str):
                continue

            subject = subject.strip()
            fact = fact.strip()

            if not subject or not fact:
                continue

            # Check whether the subject is mentioned.
            if re.search(
                rf"\b{re.escape(subject)}\b",
                content,
                re.IGNORECASE
            ):

                # Flag for human verification if the approved fact
                # is not present in the submitted content.
                if not re.search(
                    re.escape(fact),
                    content,
                    re.IGNORECASE
                ):

                    issues.append({
                        "type": "Fact Verification",
                        "severity": "Medium",
                        "message": (
                            f"Verify the claim about '{subject}'."
                        ),
                        "evidence": (
                            f"The approved fact is: {fact}. "
                            "Check that the submitted claim is consistent."
                        )
                    })

                    suggestions.append({
                        "original": subject,
                        "replacement": (
                            f"{subject}: {fact}"
                        )
                    })

    # 4. Return analysis results
    return {
        "issues": issues,
        "suggestions": suggestions,
        "overall_status": (
            "Needs Review" if issues else "Passed Initial Checks"
        )
    }

from flask import Blueprint, request, jsonify
from database import db
from models import Organisation, ReviewHistory
from services.access_control import organisation_admin_required

approvals_bp = Blueprint("approvals", __name__)


@approvals_bp.route(
    "/api/approvals/<int:review_id>",
    methods=["PUT"]
)
@organisation_admin_required
def update_approval(review_id):

    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({"error": "Valid JSON is required"}), 400

    # Accept both frontend and backend field names.
    status = data.get("approval_status", data.get("status"))
    organisation_id = data.get("organisation_id")

    if isinstance(status, str):
        status = status.strip().lower()

    status_mapping = {
        "approved": "Approved",
        "rejected": "Rejected"
    }

    if status not in status_mapping:
        return jsonify({
            "error": "Approval status must be approved or rejected"
        }), 400

    if not isinstance(organisation_id, int) or isinstance(organisation_id, bool):
        return jsonify({"error": "Valid organisation_id is required"}), 400

    try:
        organisation = db.session.get(
            Organisation,
            organisation_id
        )

        if organisation is None:
            return jsonify({"error": "Organisation not found"}), 404

        review = ReviewHistory.query.filter_by(
            id=review_id,
            organisation_id=organisation_id
        ).first()

        if review is None:
            return jsonify({"error": "Review not found"}), 404

        review.overall_status = status_mapping[status]

        db.session.commit()

        return jsonify({
            "message": "Review decision saved successfully",
            "review_id": review.id,
            "organisation_id": organisation_id,
            "status": review.overall_status
        }), 200

    except Exception:
        db.session.rollback()
        return jsonify({
            "error": "Could not save review decision"
        }), 500

from flask import Blueprint, request, jsonify
from database import db
from models import User, Organisation, OrganisationMember
from werkzeug.security import generate_password_hash, check_password_hash
from flask_jwt_extended import (
    create_access_token,
    jwt_required,
    get_jwt_identity
)

auth_bp = Blueprint("auth", __name__)

DEFAULT_ORG_ID = 1


def _session_payload(user, membership, organisation):
    return {
        "id": user.id,
        "name": user.name,
        "email": user.email,
        "role": membership.role if membership else None,
        "organisation_id": organisation.id if organisation else None,
        "organisation_name": organisation.name if organisation else None,
    }


def _membership_for(user):
    """Return (membership, organisation). Attach legacy users without a
    membership to the default organisation so they are never locked out."""
    membership = OrganisationMember.query.filter_by(
        user_id=user.id).order_by(OrganisationMember.id).first()
    if membership is None:
        organisation = db.session.get(Organisation, DEFAULT_ORG_ID)
        if organisation is None:
            return None, None
        membership = OrganisationMember(
            user_id=user.id, organisation_id=organisation.id, role="Member")
        db.session.add(membership)
        db.session.commit()
        return membership, organisation
    return membership, db.session.get(Organisation, membership.organisation_id)


# Register a new user
@auth_bp.route("/api/auth/register", methods=["POST"])
def register():

    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({"error": "Valid JSON is required"}), 400

    name = data.get("name")
    email = data.get("email")
    password = data.get("password")

    if not isinstance(name, str) or not name.strip():
        return jsonify({"error": "Name is required"}), 400

    if not isinstance(email, str) or "@" not in email:
        return jsonify({"error": "Valid email is required"}), 400

    if not isinstance(password, str) or len(password) < 8:
        return jsonify({
            "error": "Password must contain at least 8 characters"
        }), 400

    email = email.strip().lower()

    try:
        existing_user = User.query.filter_by(email=email).first()

        if existing_user:
            return jsonify({
                "error": "An account with this email already exists"
            }), 409

        # Find the demo organisation.
        organisation = db.session.get(Organisation, 1)

        if organisation is None:
            return jsonify({
                "error": "Demo organisation is not configured"
            }), 500

        # Create the user.
        user = User(
            name=name.strip(),
            email=email,
            password_hash=generate_password_hash(password)
        )

        db.session.add(user)
        db.session.flush()

        # First registered user becomes Owner.
        existing_members = OrganisationMember.query.filter_by(
            organisation_id=1
        ).count()

        role = "Owner" if existing_members == 0 else "Member"

        membership = OrganisationMember(
            user_id=user.id,
            organisation_id=1,
            role=role
        )

        db.session.add(membership)
        db.session.commit()

        access_token = create_access_token(
            identity=str(user.id)
        )

        return jsonify({
            "message": "User registered successfully",
            "access_token": access_token,
            "user": _session_payload(user, membership, organisation)
        }), 201

    except Exception:
        db.session.rollback()
        return jsonify({
            "error": "Could not register user"
        }), 500


# Login an existing user
@auth_bp.route("/api/auth/login", methods=["POST"])
def login():

    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({"error": "Valid JSON is required"}), 400

    email = data.get("email")
    password = data.get("password")

    if not isinstance(email, str) or not isinstance(password, str):
        return jsonify({"error": "Email and password are required"}), 400

    email = email.strip().lower()

    user = User.query.filter_by(email=email).first()

    if user is None or not check_password_hash(
        user.password_hash,
        password
    ):
        return jsonify({"error": "Invalid email or password"}), 401

    access_token = create_access_token(
        identity=str(user.id)
    )

    membership, organisation = _membership_for(user)

    return jsonify({
        "message": "Login successful",
        "access_token": access_token,
        "user": _session_payload(user, membership, organisation)
    }), 200


# Get current logged-in user
@auth_bp.route("/api/auth/me", methods=["GET"])
@jwt_required()
def get_current_user():

    user_id = int(get_jwt_identity())

    user = db.session.get(User, user_id)

    if user is None:
        return jsonify({"error": "User not found"}), 404

    membership, organisation = _membership_for(user)

    return jsonify(_session_payload(user, membership, organisation)), 200

from flask import Blueprint, request, jsonify
from database import db
from models import Organisation, BrandGuideline
import json
from services.access_control import (
    organisation_member_required,
    organisation_admin_required
)

guidelines_bp = Blueprint("guidelines", __name__)


# Retrieve saved guidelines

@guidelines_bp.route(
    "/api/guidelines/<int:organisation_id>",
    methods=["GET"]
)
@organisation_member_required
def get_guidelines(organisation_id):

    organisation = db.session.get(Organisation, organisation_id)

    if organisation is None:
        return jsonify({"error": "Organisation not found"}), 404

    guideline = BrandGuideline.query.filter_by(
        organisation_id=organisation_id
    ).order_by(BrandGuideline.id.desc()).first()

    if guideline is None:
        return jsonify({"error": "No guidelines found"}), 404

    return jsonify({
        "organisation_id": organisation_id,
        "tone": guideline.tone,
        "restricted_words": json.loads(guideline.restricted_words or "[]"),
        "approved_terms": json.loads(guideline.approved_terms or "[]"),
        "approved_facts": json.loads(guideline.approved_facts or "{}")
    }), 200


# Create or update guidelines

@guidelines_bp.route(
    "/api/guidelines/<int:organisation_id>",
    methods=["PUT"]
)
@organisation_admin_required
def save_guidelines(organisation_id):

    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({"error": "Valid JSON is required"}), 400

    tone = data.get("tone")
    restricted_words = data.get("restricted_words", [])
    approved_terms = data.get("approved_terms", [])
    approved_facts = data.get("approved_facts", {})

    if not isinstance(tone, str) or not tone.strip():
        return jsonify({"error": "Tone is required"}), 400

    if not isinstance(restricted_words, list):
        return jsonify({"error": "restricted_words must be a list"}), 400

    if not isinstance(approved_terms, list):
        return jsonify({"error": "approved_terms must be a list"}), 400

    if not isinstance(approved_facts, dict):
        return jsonify({"error": "approved_facts must be an object"}), 400

    if not all(isinstance(word, str) for word in restricted_words):
        return jsonify({"error": "Restricted words must contain text only"}), 400

    if not all(isinstance(term, str) for term in approved_terms):
        return jsonify({"error": "Approved terms must contain text only"}), 400

    if not all(
        isinstance(key, str) and isinstance(value, str)
        for key, value in approved_facts.items()
    ):
        return jsonify({"error": "Approved facts must contain text keys and values"}), 400

    try:
        organisation = db.session.get(Organisation, organisation_id)

        if organisation is None:
            return jsonify({"error": "Organisation not found"}), 404

        guideline = BrandGuideline.query.filter_by(
            organisation_id=organisation_id
        ).order_by(BrandGuideline.id.desc()).first()

        if guideline is None:
            guideline = BrandGuideline(
                organisation_id=organisation_id
            )
            db.session.add(guideline)

        guideline.tone = tone.strip()
        guideline.restricted_words = json.dumps(restricted_words)
        guideline.approved_terms = json.dumps(approved_terms)
        guideline.approved_facts = json.dumps(approved_facts)

        db.session.commit()

        return jsonify({
            "message": "Guidelines saved successfully",
            "organisation_id": organisation_id
        }), 200

    except Exception:
        db.session.rollback()
        return jsonify({"error": "Could not save guidelines"}), 500

            try:

            except (json.JSONDecodeError, TypeError):
                issues = []

                suggestions = []

                "review_id": item.id,
                "submitted_content": item.submitted_content,
                "issues": issues,
                "suggestions": suggestions,
                "overall_status": item.overall_status,
                "created_at": (
                    item.created_at.isoformat()
                    if item.created_at else None
                )
            })

        return jsonify({
            "organisation_id": organisation_id,
            "total": len(reviews),
            "reviews": reviews,
            "history": reviews
        }), 200

    except Exception:

from flask import Blueprint, request, jsonify

@review_bp.route("/api/analyze", methods=["POST"])
@organisation_member_required
def analyze():

    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({"error": "Valid JSON is required"}), 400

    organisation_id = data.get("organisation_id")
    content = data.get("content")

    if not isinstance(organisation_id, int) or isinstance(organisation_id, bool):
        return jsonify({"error": "Valid organisation_id is required"}), 400

    if not isinstance(content, str) or not content.strip():
        return jsonify({"error": "Content cannot be empty"}), 400

    try:
        organisation = db.session.get(Organisation, organisation_id)

        if organisation is None:
            return jsonify({"error": "Organisation not found"}), 404

        guideline = BrandGuideline.query.filter_by(
            organisation_id=organisation_id
        ).order_by(BrandGuideline.id.desc()).first()

        if guideline is None:
            return jsonify({"error": "No brand guidelines found"}), 404

        guidelines = {
            "tone": guideline.tone,
            "restricted_words": json.loads(guideline.restricted_words),
            "approved_terms": json.loads(guideline.approved_terms),
            "approved_facts": json.loads(guideline.approved_facts)
        }

        from analyzer import analyze_content

        result = analyze_content(content, guidelines)

        required_keys = ["issues", "suggestions", "overall_status"]

        if not isinstance(result, dict):
            raise ValueError("Invalid analyzer response")

        if not all(key in result for key in required_keys):
            raise ValueError("Missing analyzer response fields")

        if not isinstance(result["issues"], list):
            raise ValueError("Issues must be a list")

        if not isinstance(result["suggestions"], list):
            raise ValueError("Suggestions must be a list")

        if not isinstance(result["overall_status"], str):
            raise ValueError("Invalid review status")

        # Convert analyzer output into the frontend's expected format.
        frontend_issues = []

        for index, issue in enumerate(result["issues"]):
            if not isinstance(issue, dict):
                continue

            suggestion_data = (
                result["suggestions"][index]
                if index < len(result["suggestions"])
                and isinstance(result["suggestions"][index], dict)
                else {}
            )

            frontend_issues.append({
                "id": index + 1,
                "category": issue.get("type", "Issue"),
                "severity": str(issue.get("severity", "low")).lower(),
                "original_text": (
                    suggestion_data.get("original")
                    or issue.get("original_text")
                    or issue.get("evidence", "")
                ),
                "explanation": (
                    issue.get("evidence")
                    or issue.get("message", "")
                ),
                "suggestion": (
                    suggestion_data.get("replacement")
                    or issue.get("suggestion", "")
                )
            })

        # Save the frontend-compatible issues.
        review = ReviewHistory(
            organisation_id=organisation_id,
            submitted_content=content,
            issues=json.dumps(frontend_issues),
            suggestions=json.dumps(result["suggestions"]),
            overall_status=result["overall_status"]
        )

        db.session.add(review)
        db.session.commit()

        return jsonify({
            "review_id": review.id,
            "organisation_id": organisation_id,
            "issues": frontend_issues,
            "suggestions": result["suggestions"],
            "overall_status": result["overall_status"]
        }), 200

    except ModuleNotFoundError as error:
        db.session.rollback()

        if error.name == "analyzer":
            return jsonify({
                "error": "Analysis engine is not integrated yet"
            }), 503

        return jsonify({
            "error": "An analyzer dependency is missing"
        }), 500

    except Exception:
        db.session.rollback()

        return jsonify({
            "error": "Content analysis failed"
        }), 500

from functools import wraps

from flask import jsonify, request
from flask_jwt_extended import jwt_required, get_jwt_identity

from database import db
from models import Organisation, OrganisationMember


def get_organisation_id(kwargs):
    organisation_id = kwargs.get("organisation_id")

    if organisation_id is None:
        data = request.get_json(silent=True) or {}
        organisation_id = data.get("organisation_id")

    if isinstance(organisation_id, bool):
        return None

    try:
        organisation_id = int(organisation_id)
    except (TypeError, ValueError):
        return None

    if organisation_id <= 0:
        return None

    return organisation_id


def organisation_member_required(function):

    @wraps(function)
    @jwt_required()
    def wrapper(*args, **kwargs):

        organisation_id = get_organisation_id(kwargs)

        if organisation_id is None:
            return jsonify({
                "error": "Valid organisation_id is required"
            }), 400

        organisation = db.session.get(Organisation, organisation_id)

        if organisation is None:
            return jsonify({
                "error": "Organisation not found"
            }), 404

        user_id = int(get_jwt_identity())

        membership = OrganisationMember.query.filter_by(
            user_id=user_id,
            organisation_id=organisation_id
        ).first()

        if membership is None:
            return jsonify({
                "error": "You are not a member of this organisation"
            }), 403

        return function(*args, **kwargs)

    return wrapper


def organisation_admin_required(function):

    @wraps(function)
    @jwt_required()
    def wrapper(*args, **kwargs):

        organisation_id = get_organisation_id(kwargs)

        if organisation_id is None:
            return jsonify({
                "error": "Valid organisation_id is required"
            }), 400

        organisation = db.session.get(Organisation, organisation_id)

        if organisation is None:
            return jsonify({
                "error": "Organisation not found"
            }), 404

        user_id = int(get_jwt_identity())

        membership = OrganisationMember.query.filter_by(
            user_id=user_id,
            organisation_id=organisation_id
        ).first()

        if membership is None:
            return jsonify({
                "error": "You are not a member of this organisation"
            }), 403

        if membership.role not in ["Owner", "Admin"]:
            return jsonify({
                "error": "You do not have permission to perform this action"
            }), 403

        return function(*args, **kwargs)

    return wrapper

  async function checkAboutHealth() {

    if (!badge || !details) return;

    const start = performance.now();

      if (res.ok) {
        badge.className = "badge resolved";
        badge.textContent = `Online (${latency}ms)`;
        details.textContent = `All governance inference nodes operational. Response latency is optimal.`;
      } else {
        throw new Error("Server error");
      }
    } catch (err) {
      badge.className = "badge high";
      badge.textContent = "Offline / Mock Mode";
      details.textContent = `Unable to reach live backend at /api/health. Operating securely on local fallback fallback mode.`;
    }
  }

  document.addEventListener("DOMContentLoaded", checkAboutHealth);
})();

const ApiClient = (() => {
  const SESSION_KEY = "brandtrace.session";
  const BASE_KEY = "brandtrace.apiBase";
  const MOCK_KEY = "brandtrace.forceMock";

  const isLocal = window.location.hostname === "localhost" || window.location.hostname === "127.0.0.1";
  const defaultBase = isLocal ? "http://127.0.0.1:5000" : "https://brandtrace-jhli.onrender.com";

  const cfg = {
    base: localStorage.getItem(BASE_KEY) || defaultBase,
    prefix: "/api",
    timeoutMs: 8000,
    healthMs: 2000,
    get forceMock() { return localStorage.getItem(MOCK_KEY) === "1"; },
  };

  class ApiError extends Error {
    constructor(message, status) {
      super(message);
      this.name = "ApiError";
      this.status = status;
    }
  }

  // ---------- session ----------
  function getSession() {
    try { return JSON.parse(localStorage.getItem(SESSION_KEY)) || {}; }
    catch (e) { return {}; }
  }
  function saveSession(s) { localStorage.setItem(SESSION_KEY, JSON.stringify(s)); }
  function clearSession() {
    try { localStorage.removeItem(SESSION_KEY); } catch (e) {}
  }

  // A real JWT has exactly 3 dot-separated segments. Anything else
  // ("mock-jwt-token-for-demo-mode", "demo-judge-token") makes
  // flask-jwt-extended answer 422/401, so we never send it.
  const looksLikeJwt = t => typeof t === "string" && t.split(".").length === 3;

  function token() {
    const s = getSession();
    const t = s.access_token || s.token || null;
    return looksLikeJwt(t) ? t : null;
  }

  function hasValidSession() { return !!token(); }

  function getOrgId() {
    const s = getSession();
    const id = Number(s.org_id || s.organisation_id);
    return Number.isInteger(id) && id > 0 ? id : 1;
  }

  function redirectToLogin(reason) {
    clearSession();
    if (!/auth\.html$|landing\.html$/.test(location.pathname)) {
      const q = reason ? "?reason=" + encodeURIComponent(reason) : "";
      location.href = "auth.html" + q;
    }
  }

  // ---------- core request ----------
  // opts.auth === false  -> do not attach the Authorization header (login/register)
  async function request(method, path, body, timeout = cfg.timeoutMs, opts = {}) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), timeout);
    const headers = { Accept: "application/json" };
    if (body !== undefined) headers["Content-Type"] = "application/json";

    const authToken = opts.auth === false ? null : token();
    if (authToken) headers["Authorization"] = "Bearer " + authToken;

    try {
      const res = await fetch(cfg.base + cfg.prefix + path, {
        method, headers, signal: ctrl.signal,
        body: body !== undefined ? JSON.stringify(body) : undefined,
      });

      if (!res.ok) {
        let errorBody = {};
        try { errorBody = await res.json(); } catch (e) {}
        const msg = errorBody.error || errorBody.message || errorBody.msg || `HTTP error ${res.status}`;

        // 401 (and 422 from any stale JWT) on a protected call = session is dead.
        if ((res.status === 401 || res.status === 422) && opts.auth !== false) {
          redirectToLogin("expired");
        }
        throw new ApiError(msg, res.status);
      }
      return await res.json();

    } catch (err) {
      if (err.name === "AbortError") throw new ApiError("Request timed out connecting to server", 0);
      // fetch() network failure (server down / CORS / wrong port) is a TypeError
      if (err instanceof TypeError) throw new ApiError("Cannot reach server at " + cfg.base, 0);
      throw err;
    } finally {
      clearTimeout(timer);
    }
  }

  const isOffline = err => !!err && err.status === 0;

  // ---------- guidelines shape helpers ----------
  // Local (Storage) shape  <->  backend shape
  function toBackendGuidelines(g) {
    let facts = g.approved_facts;
    if (!facts || typeof facts !== "object" || Array.isArray(facts)) facts = {};
    return {
      tone: (g.tone || "").trim() || "Confident, clear and customer-friendly.",
      restricted_words: (g.restricted_words || []).filter(w => typeof w === "string" && w.trim()),
      approved_terms: (g.approved_terms || []).filter(w => typeof w === "string" && w.trim()),
      approved_facts: facts,
    };
  }

  function normalizeIssue(i, n) {
    const sev = String(i.severity || "low").toLowerCase();
    return {
      id: i.id || "iss-" + n,
      category: i.category || i.type || "Issue",
      severity: ["high", "medium", "low"].includes(sev) ? sev : "low",
      original_text: i.original_text || i.original || i.excerpt || "",
      explanation: i.explanation || i.reason || "",
      suggestion: i.suggestion || i.suggested_edit || "",
    };
  }

  async function checkHealth() {
    if (cfg.forceMock) { window.setBackendStatus?.(false); return false; }
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), cfg.healthMs);
    try {
      const res = await fetch(cfg.base + "/api/health", { signal: ctrl.signal });
      if (!res.ok) throw new Error("health failed");
      window.setBackendStatus?.(true);
      return true;
    } catch (e) {
      window.setBackendStatus?.(false);
      return false;
    } finally { clearTimeout(timer); }
  }

  async function analyzeContent(content, guidelines) {
    const orgId = getOrgId();

    if (cfg.forceMock) {
      await new Promise(r => setTimeout(r, 600));
      return { ...SampleData.mockAnalyze(content, guidelines), source: "mock" };
    }

    const send = () => request("POST", "/analyze", { organisation_id: orgId, content });

    let data;
    try {
      try {
        data = await send();
      } catch (err) {
        // Org has no guidelines on the server yet: push the local ones once, retry.
        if (err.status === 404 && /guidelines/i.test(err.message)) {
          await request("PUT", `/guidelines/${orgId}`, toBackendGuidelines(guidelines));
          data = await send();
        } else { throw err; }
      }
    } catch (err) {
      window.setBackendStatus?.(!isOffline(err));
      throw err;
    }

    window.setBackendStatus?.(true);
    const list = data.issues || (data.data && data.data.issues) || [];
    return {
      issues: list.map(normalizeIssue),
      source: "live",
      review_id: data.review_id,
      overall_status: data.overall_status,
    };
  }

  setInterval(checkHealth, 30000);

  const endpoints = {
    getGuidelines: (orgId = getOrgId()) => request("GET", `/guidelines/${orgId}`),
    saveGuidelines: (payload, orgId = getOrgId()) => request("PUT", `/guidelines/${orgId}`, payload),
    fetchDashboard: (orgId = getOrgId()) => request("GET", `/organisations/${orgId}/dashboard`),
    fetchHistory: (orgId = getOrgId()) => request("GET", `/organisations/${orgId}/history`),
    fetchOrganisation: (orgId = getOrgId()) => request("GET", `/organisations/${orgId}`),
    updateOrganisation: (payload, orgId = getOrgId()) => request("PUT", `/organisations/${orgId}`, payload),
    addMember: (email, orgId = getOrgId()) => request("POST", `/organisations/${orgId}/members`, { email }),
    decideReview: (reviewId, status, orgId = getOrgId()) =>
      request("PUT", `/approvals/${reviewId}`, { organisation_id: orgId, approval_status: status }),
  };

  return {
    request, analyzeContent, checkHealth, ApiError, getOrgId, getSession, saveSession,
    clearSession, hasValidSession, isOffline, toBackendGuidelines, redirectToLogin,
    // back-compat aliases (app.js used these without `.endpoints`)
    getGuidelines: endpoints.getGuidelines,
    saveGuidelines: endpoints.saveGuidelines,
    get base() { return cfg.base; },
    get isMock() { return cfg.forceMock; },
    setMock: v => localStorage.setItem(MOCK_KEY, v ? "1" : "0"),
    endpoints,
  };
})();

  async function initGuidelines() {
    const box = $("chip-box");
    const input = $("chip-input");
    const severitySelect = $("rule-severity");
    const status = $("save-status");
    const form = $("guidelines-form");
    const sandboxInput = $("sandbox-input");
    const sandboxFeedback = $("sandbox-feedback");
    const toneInput = $("tone");
    const specsInput = $("specs");

    if (!box || !input || !severitySelect || !form || !toneInput || !specsInput) return;

    const localSaved = Storage.load();
    let restrictedRules = [];
    let approvedTerms = [];   // round-tripped so saving never wipes them

    const specLines = () =>
      specsInput.value.split("\n").map(l => l.trim()).filter(Boolean);

    const factsToLines = facts => {
      if (Array.isArray(facts)) {
        return facts.map(item =>
          typeof item === "string" ? item
            : `${item.subject || "General"}: ${item.value || ""}`.trim());
      }
      if (facts && typeof facts === "object") {
        return Object.entries(facts).map(([k, v]) => `${k}: ${v}`);
      }
      return [];
    };

    const linesToFacts = () => {
      const facts = {};
      specLines().forEach(line => {
        const i = line.indexOf(":");
        const subject = i >= 0 ? line.slice(0, i).trim() : "General";
        const value = i >= 0 ? line.slice(i + 1).trim() : line;
        if (subject && value) facts[subject] = value;
      });
      return facts;
    };

    // Per-term severity only exists in the browser; the backend stores plain words.
    const localSeverity = {};
    (localSaved.restricted_terms || []).forEach(r => {
      if (r && r.term) localSeverity[r.term.toLowerCase()] = r.severity || "medium";
    });

    const loadRules = rules =>
      (Array.isArray(rules) ? rules : [])
        .map(rule => typeof rule === "string"
          ? { term: rule, severity: localSeverity[rule.toLowerCase()] || "medium" }
          : { term: rule.term || "", severity: rule.severity || "medium" })
        .filter(rule => rule.term);

    const persistLocal = () => Storage.save({
      tone: toneInput.value.trim(),
      restricted_terms: restrictedRules,
      restricted_words: restrictedRules.map(r => r.term),
      approved_terms: approvedTerms,
      approved_facts: linesToFacts(),
    });

    // ---- Load from backend (fallback: this browser's copy) ----
    try {
      const data = await ApiClient.endpoints.getGuidelines();
      restrictedRules = loadRules(data.restricted_words);
      approvedTerms = Array.isArray(data.approved_terms) ? data.approved_terms : [];
      toneInput.value = data.tone || "";
      specsInput.value = factsToLines(data.approved_facts).join("\n");
      persistLocal();
      if (status) status.textContent = "Guidelines loaded from server";
    } catch (error) {
      restrictedRules = loadRules(localSaved.restricted_terms || localSaved.restricted_words);
      approvedTerms = localSaved.approved_terms || [];
      toneInput.value = localSaved.tone || "";
      specsInput.value = factsToLines(localSaved.approved_facts).join("\n");
      if (status) {
        status.textContent =
          error.status === 404 ? "No saved server guidelines yet. Edit and save to create them."
          : error.status === 0 ? "Server offline. Showing this browser's saved copy."
          : `Could not load server guidelines (${error.message}). Showing this browser's saved copy.`;
      }
      console.warn("Could not load guidelines from backend:", error);
    }

    // ---- Sandbox ----
    function updateSandbox() {
      if (!sandboxInput || !sandboxFeedback) return;
      const sample = sandboxInput.value.toLowerCase();
      if (!sample) { sandboxFeedback.textContent = ""; sandboxFeedback.className = ""; return; }
      const matched = restrictedRules.find(r => sample.includes(r.term.toLowerCase()));
      if (matched) {
        sandboxFeedback.className = "sandbox-fail";
        sandboxFeedback.textContent =
          `❌ Blocked: Contains restricted term "${matched.term}" (${matched.severity.toUpperCase()} severity).`;
      } else {
        sandboxFeedback.className = "sandbox-pass";
        sandboxFeedback.textContent = "✔ Passed: No restricted terms detected in test text.";
      }
    }

    function updateCount() {
      const count = $("rule-count");
      if (count) count.textContent = `${restrictedRules.length} words · ${specLines().length} facts`;
    }

    function renderChips() {
      box.querySelectorAll(".chip").forEach(chip => chip.remove());
      restrictedRules.forEach((rule, index) => {
        const chip = document.createElement("span");
        chip.className = `chip sev-${rule.severity}`;
        chip.innerHTML = `
          ${esc(rule.term)}
          <small>(${esc(rule.severity)})</small>
          <button type="button" data-i="${index}" aria-label="Remove ${esc(rule.term)}">×</button>`;
        box.appendChild(chip);
      });
      updateCount();
      updateSandbox();
    }

    function addRule(term, severity) {
      const value = term.trim().replace(/,$/, "");
      input.value = "";
      if (!value) return;
      if (restrictedRules.some(r => r.term.toLowerCase() === value.toLowerCase())) return;
      restrictedRules.push({ term: value, severity: severity || "medium" });
      renderChips();
    }

    input.addEventListener("keydown", event => {
      if (event.key === "Enter" || event.key === ",") {
        event.preventDefault();
        addRule(input.value, severitySelect.value);
      }
    });

    box.addEventListener("click", event => {
      const removeButton = event.target.closest("button[data-i]");
      if (!removeButton) return;
      restrictedRules.splice(Number(removeButton.dataset.i), 1);
      renderChips();
    });

    sandboxInput?.addEventListener("input", updateSandbox);
    specsInput.addEventListener("input", updateCount);

    // ---- Save ----
    form.addEventListener("submit", async event => {
      event.preventDefault();
      if (input.value.trim()) addRule(input.value, severitySelect.value);

      const payload = {
        tone: toneInput.value.trim(),
        restricted_words: restrictedRules.map(r => r.term),
        approved_terms: approvedTerms,
        approved_facts: linesToFacts(),
      };

      if (!payload.tone) {
        setError(toneInput, "Tone is required.");
        toneInput.focus();
        return;
      }
      clearError(toneInput);

      const saveButton = form.querySelector('[type="submit"]');
      if (saveButton) saveButton.disabled = true;
      if (status) status.textContent = "Saving guidelines to server…";

      try {
        await ApiClient.endpoints.saveGuidelines(payload);
        persistLocal();
        if (status) status.textContent = "Saved to server ✔ · " + timeLabel();
        notify("Guidelines saved successfully.", "success");
      } catch (error) {
        const msg = error.status === 403
          ? "Only organisation Owners/Admins can edit guidelines."
          : error.message || "Backend error";
        if (status) status.textContent = `Save failed: ${msg}`;
        notify(`Guidelines were not saved: ${msg}`, "error");
      } finally {
        if (saveButton) saveButton.disabled = false;
      }
    });

    renderChips();
  }

(function () {
  const { esc } = Utils;
  const $ = id => document.getElementById(id);

  // Group any backend status string into one of four buckets.
  // The backend can send "Needs Review", "Passed Initial Checks", "Approved", "Rejected" or "Pending".
  function bucket(status) {
    const s = String(status || "").toLowerCase();
    if (s.includes("reject")) return "rejected";
    if (s.includes("approve")) return "approved";
    if (s.includes("pass")) return "passed";
    return "attention";
  }

  const badgeClass = b =>
    b === "approved" || b === "passed" ? "resolved" : b === "rejected" ? "neutral" : "high";

  function fmtDate(value) {
    const d = value ? new Date(value) : null;
    return d && !isNaN(d) ? d.toLocaleString([], { dateStyle: "medium", timeStyle: "short" }) : "—";
  }

  function excerpt(text) {
    const t = String(text || "").trim();
    if (!t) return "Draft copy";
    return t.length > 60 ? t.slice(0, 60) + "…" : t;
  }

  function setNotice(html) {
    $("dash-notice").innerHTML = html
      ? `<section class="card" style="margin-bottom: 16px;">${html}</section>`
      : "";
  }

  function renderStats(list) {
    const counts = { approved: 0, rejected: 0, passed: 0, attention: 0 };
    list.forEach(r => counts[bucket(r.overall_status)]++);
    $("dash-total").textContent = list.length;
    $("dash-attention").textContent = counts.attention;
    $("dash-approved").textContent = counts.approved;
    $("dash-rejected").textContent = counts.rejected;
  }

  function renderRecent(list) {
    const tbody = $("dash-recent");

    if (!list.length) {
      tbody.innerHTML = `
        <tr><td colspan="5" style="text-align: center; padding: 32px; color: var(--muted);">
          No audits yet. <a href="content_analysis.html">Analyse your first piece of copy</a> and it will appear here.
        </td></tr>`;
      return;
    }

    tbody.innerHTML = list.slice(0, 5).map(item => {
      const id = item.id || item.review_id || "—";
      const status = item.overall_status || "Needs Review";
      const count = Array.isArray(item.issues) ? item.issues.length : 0;
      return `
        <tr>
          <td><strong>#${esc(id)}</strong></td>
          <td style="color: var(--muted); white-space: nowrap;">${esc(fmtDate(item.created_at))}</td>
          <td>${esc(excerpt(item.submitted_content))}</td>
          <td class="nowrap-cell"><span class="badge neutral">${count} ${count === 1 ? "issue" : "issues"}</span></td>
          <td class="nowrap-cell"><span class="badge ${badgeClass(bucket(status))}">${esc(status)}</span></td>
        </tr>`;
    }).join("");
  }

  async function load() {
    const session = ApiClient.getSession();
    const first = String(session.name || "").trim().split(/\s+/)[0];
    if (first) $("dash-title").textContent = `Welcome back, ${first}`;

    const [dash, hist] = await Promise.allSettled([
      ApiClient.endpoints.fetchDashboard(orgId),
      ApiClient.endpoints.fetchHistory(orgId),
    ]);

    // History drives the numbers: the backend dashboard counts only the literal
    // status "Pending", which the analyser never produces.
    if (hist.status === "fulfilled") {
      const data = hist.value;
      const list = Array.isArray(data) ? data : data.reviews || data.history || [];
      renderStats(list);
      renderRecent(list);
    } else {

      $("dash-recent").innerHTML = `
        <tr><td colspan="5" style="text-align: center; padding: 32px; color: var(--muted);">

const AppState = (() => {
  let listeners = [];
  let state = {
    workbench: { content: "", issues: [], source: null },
    guidelines: { tone: "", words: [], specs: [] },
    settings: {},
  };

  function get() {
    return state;
  }

  function set(newState) {
    state = { ...state, ...newState };
    listeners.forEach(fn => fn(state));
  }

  function subscribe(fn) {
    listeners.push(fn);
    return () => {
      listeners = listeners.filter(l => l !== fn);
    };
  }

  return { get, set, subscribe };
})();

const Storage = (() => {
  const KEY = "brandtrace.guidelines.v1";
  const DEFAULTS = {
    tone: "Confident, clear and customer-friendly. Avoid superlatives and unverifiable claims.",
    restricted_words: ["Cheapest", "Best in class", "Guaranteed"],
    // Store approved facts as a key-value dictionary to match backend expectations
    approved_facts: {
      "NovaBook Pro battery life": "Up to 12 hours",
      "NovaBook Pro display": "14-inch 2.8K OLED",
      "NovaBook Pro weight": "1.3 kg"
    },
  };

  function load() {
    try {
      const raw = localStorage.getItem(KEY);
      if (!raw) return structuredClone(DEFAULTS);
      const parsed = JSON.parse(raw);
      
      // Backward compatibility if older storage used an array for approved_specs
      if (Array.isArray(parsed.approved_specs)) {
        const convertedFacts = {};
        parsed.approved_specs.forEach(line => {
          const parts = line.split(":");
          if (parts.length >= 2) {
            convertedFacts[parts[0].trim()] = parts.slice(1).join(":").trim();
          } else if (line.trim()) {
            convertedFacts[line.trim()] = "Verified";
          }
        });
        parsed.approved_facts = convertedFacts;
        delete parsed.approved_specs;
      }

      return { ...DEFAULTS, ...parsed };
    } catch (e) { 
      return structuredClone(DEFAULTS); 
    }
  }

  function save(g) {
    try { localStorage.setItem(KEY, JSON.stringify(g)); } catch (e) {}
  }
  
  function reset() { 
    localStorage.removeItem(KEY); 
  }

  return { load, save, reset, DEFAULTS };
})();

const Utils = (() => {
  const esc = s => String(s).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  const debounce = (fn, ms = 400) => { let t; return (...a) => { clearTimeout(t); t = setTimeout(() => fn(...a), ms); }; };
  const timeLabel = () => new Date().toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" });

  const store = {
    get(k, d) { try { const v = localStorage.getItem(k); return v ? JSON.parse(v) : d; } catch (e) { return d; } },
    set(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); return true; } catch (e) { return false; } },
    del(k) { try { localStorage.removeItem(k); } catch (e) {} },
  };

  function toast(msg, type = "success") {
    let t = document.getElementById("toast");
    if (!t) { t = document.createElement("div"); t.id = "toast"; document.body.appendChild(t); }
    t.className = `toast ${type} show`; t.textContent = msg;
    clearTimeout(t._h); t._h = setTimeout(() => t.classList.remove("show"), 2600);
  }

  // Inline validation: puts a message directly under the field (or chip box).
  function setError(el, msg) {
    el.classList.add("invalid"); el.setAttribute("aria-invalid", "true");
    let n = el.nextElementSibling;
    if (!n || !n.classList.contains("field-error")) {
      n = document.createElement("div"); n.className = "field-error"; n.setAttribute("role", "alert"); el.after(n);
    }
    n.textContent = msg;
  }
  function clearError(el) {
    el.classList.remove("invalid"); el.removeAttribute("aria-invalid");
    const n = el.nextElementSibling;
    if (n && n.classList.contains("field-error")) n.remove();
  }
  return { esc, debounce, timeLabel, store, toast, setError, clearError };
})();
